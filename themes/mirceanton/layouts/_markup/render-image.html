{{- /* Images resolve from the page bundle first, then from assets/. They get real dimensions and lazy loading; the theme's script opens them full size on click. Photos and screenshots are served as WebP copies no wider than 1600px, with an 800px one for small screens and a copy of up to 2400px for the zoom view, so a huge original never reaches the reader. An image that is already a WebP within 1600px is served as it is. SVG and GIF pass through untouched. A caption comes from an {caption="..."} attribute, or from an _italic line_ right below. */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- $res := false -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $res = or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- end -}} {{- $shown := false -}} {{- $dims := "" -}} {{- $srcset := "" -}} {{- $zoom := "" -}} {{- with $res -}} {{- if in (slice "jpeg" "png" "webp" "tiff" "bmp") .MediaType.SubType -}} {{- $shown = . -}} {{- if or (gt .Width 1600) (ne .MediaType.SubType "webp") -}} {{- $shown = .Resize (printf "%dx webp q82" (math.Min .Width 1600 | int)) -}} {{- end -}} {{- if gt .Width 1600 -}} {{- $zoom = (.Resize (printf "%dx webp q82" (math.Min .Width 2400 | int))).RelPermalink -}} {{- end -}} {{- if gt $shown.Width 1000 -}} {{- $small := .Resize "800x webp q82" -}} {{- $srcset = printf "%s 800w, %s %dw" $small.RelPermalink $shown.RelPermalink $shown.Width -}} {{- end -}} {{- $src = $shown.RelPermalink -}} {{- else -}} {{- /* Asking for the permalink is what publishes a file, so the original is only asked for here, where it is the one being served. */ -}} {{- $src = .RelPermalink -}} {{- if eq .MediaType.SubType "gif" -}} {{- $dims = printf ` width="%d" height="%d"` .Width .Height | safeHTMLAttr -}} {{- end -}} {{- end -}} {{- end -}} {{- with .Attributes.caption }}{{ . }}{{ end -}}
