{{- /* Images resolve from the page bundle first, then from assets/. They get real dimensions and lazy loading; the theme's script opens them full size on click. A caption comes from an {caption="..."} attribute, or from an _italic line_ right below. */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := $u.String -}} {{- $res := false -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $res = or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- with $res }}{{ $src = .RelPermalink }}{{ end -}} {{- end -}} {{- with .Attributes.caption }}{{ . }}{{ end -}}
